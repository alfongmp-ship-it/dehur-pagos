    const base = fechaFn.slice(2); // yymmdd
    return (base + String(idx + 1).padStart(3,'0')).slice(0,20);
  };

  // Encabezados estilo BBVA Pagos Mixtos
  const data = [
    ['Numero de Registros==>>','',exportables.length],

      (d.concepto || '').substring(0, 40),

      '',
      '',
      ''
